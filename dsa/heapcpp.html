📊 Max Heap – Insert & Display

Max Heap Insert Array
Problem: Sanju collects participant scores and maintains a leaderboard using a max heap. Insert a new score and display the updated heap.
C++ 
// Max Heap – Insert & Display (Array‑based)
                            #include <iostream>
                            using namespace std;

                            const int MAX_SIZE = 16;

                            void swap(int& a, int& b) {
                            int temp = a; a = b; b = temp;
                            }

                            void maxHeapify(int arr[], int n, int i) {
                            int largest = i;
                            int left = 2*i + 1;
                            int right = 2*i + 2;
                            if (left < n && arr[left] > arr[largest]) largest = left;
                            if (right < n && arr[right] > arr[largest]) largest = right;
                            if (largest != i) {
                            swap(arr[i], arr[largest]);
                            maxHeapify(arr, n, largest);
                            }
                            }

                            void buildMaxHeap(int arr[], int n) {
                            for (int i = n/2 - 1; i >= 0; i--)
                            maxHeapify(arr, n, i);
                            }

                            void insertIntoMaxHeap(int arr[], int& n, int newScore) {
                            arr[n] = newScore; n++;
                            int i = n - 1;
                            while (i > 0 && arr[(i-1)/2] < arr[i]) {
                            swap(arr[i], arr[(i-1)/2]);
                            i = (i-1)/2;
                            }
                            }

                            int main() {
                            int scores[MAX_SIZE], n, d;
                            cin >> n;
                            for (int i=0; i<n; i++) cin >> scores[i];
                            cin >> d;
                            buildMaxHeap(scores, n);
                            insertIntoMaxHeap(scores, n, d);
                            for (int i=0; i<n; i++)
                            cout << scores[i] << (i<n-1 ? " " : "");
                            return 0;
                            }
💡 Key idea: Build a max heap from initial scores, then insert the new score at the end and bubble it up until the heap property is restored. Time: O(log n) for insertion.
📥 Input
5
3 1 4 2 5
6
📤 Output
6 4 5 1 2 3

🔢 Min Heap – Even & Odd Sums

Min Heap Sum Array
Problem: Aiden organizes book categories using a min heap. Build the heap, display it, and calculate the sum of even and odd numbers separately.
C++ 
// Min Heap – Even / Odd Sums
                            #include <iostream>
                            using namespace std;

                            const int MAX_SIZE = 11;

                            void swap(int& a, int& b) { int t = a; a = b; b = t; }

                            void minHeapify(int arr[], int n, int i) {
                            int smallest = i, left = 2*i+1, right = 2*i+2;
                            if (left < n && arr[left] < arr[smallest]) smallest = left;
                            if (right < n && arr[right] < arr[smallest]) smallest = right;
                            if (smallest != i) {
                            swap(arr[i], arr[smallest]);
                            minHeapify(arr, n, smallest);
                            }
                            }

                            void buildMinHeap(int arr[], int n) {
                            for (int i = n/2 - 1; i >= 0; i--) minHeapify(arr, n, i);
                            }

                            int sumEven(int arr[], int n) {
                            int s = 0;
                            for (int i=0; i<n; i++) if (arr[i] % 2 == 0) s += arr[i];
                            return s;
                            }

                            int sumOdd(int arr[], int n) {
                            int s = 0;
                            for (int i=0; i<n; i++) if (arr[i] % 2 != 0) s += arr[i];
                            return s;
                            }

                            int main() {
                            int arr[MAX_SIZE], n;
                            cin >> n;
                            for (int i=0; i<n; i++) cin >> arr[i];
                            buildMinHeap(arr, n);
                            cout << "Min Heap: ";
                            for (int i=0; i<n; i++) cout << arr[i] << (i<n-1?" ":"");
                            cout << endl;
                            cout << "Sum of even elements: " << sumEven(arr, n) << endl;
                            cout << "Sum of odd elements: " << sumOdd(arr, n) << endl;
                            return 0;
                            }
💡 Key idea: Min heap ensures the smallest element is at the root. After building, we simply traverse the array to sum even and odd values.
📥 Input
5
2 4 1 5 9
📤 Output
Min Heap: 1 4 2 5 9
Sum of even elements: 6
Sum of odd elements: 15

🔬 Min Heap – Delete Smallest Prime

Min Heap Delete Prime Array
Problem: Mia manages a min heap. Insert elements, display the heap, then delete the smallest prime number and display the updated heap.
C++ 
// Min Heap – Delete Smallest Prime
                            #include <iostream>
                            using namespace std;

                            const int MAX_SIZE = 100;

                            bool isPrime(int num) {
                            if (num <= 1) return false;
                            for (int i=2; i*i <= num; i++)
                            if (num % i == 0) return false;
                            return true;
                            }

                            void swap(int& a, int& b) { int t = a; a = b; b = t; }

                            void minHeapify(int arr[], int n, int i) {
                            int s = i, l = 2*i+1, r = 2*i+2;
                            if (l < n && arr[l] < arr[s]) s = l;
                            if (r < n && arr[r] < arr[s]) s = r;
                            if (s != i) { swap(arr[i], arr[s]); minHeapify(arr, n, s); }
                            }

                            void buildMinHeap(int arr[], int n) {
                            for (int i = n/2 - 1; i >= 0; i--) minHeapify(arr, n, i);
                            }

                            void deleteSmallestPrime(int arr[], int& n) {
                            int idx = -1;
                            for (int i=0; i<n; i++) {
                            if (isPrime(arr[i]) && (idx == -1 || arr[i] < arr[idx])) idx = i;
                            }
                            if (idx == -1) return;
                            arr[idx] = arr[n-1]; n--;
                            minHeapify(arr, n, idx);
                            }

                            int main() {
                            int heap[MAX_SIZE], n;
                            cin >> n;
                            for (int i=0; i<n; i++) cin >> heap[i];
                            buildMinHeap(heap, n);
                            cout << "Min Heap: ";
                            for (int i=0; i<n; i++) cout << heap[i] << (i<n-1?" ":"");
                            cout << endl;
                            deleteSmallestPrime(heap, n);
                            cout << "Updated Min Heap: ";
                            for (int i=0; i<n; i++) cout << heap[i] << (i<n-1?" ":"");
                            return 0;
                            }
💡 Key idea: Scan the heap to find the smallest prime, replace it with the last element, reduce size, and heapify from that index to restore the min-heap property.
📥 Input
5
4 3 2 5 6
📤 Output
Min Heap: 2 4 3 5 6
Updated Min Heap: 3 4 6 5

🌿 Max Heap – Delete Root & Leaf Sum

Max Heap Delete Leaf Sum Array
Problem: Ram manages a max heap. Insert numbers, print the heap, delete the root, print the updated heap, and calculate the sum of all remaining leaf nodes.
C++ 
// Max Heap – Delete Root + Leaf Sum
                            #include <iostream>
                            using namespace std;

                            const int MAX_SIZE = 100;

                            void swap(int arr[], int a, int b) { int t = arr[a]; arr[a] = arr[b]; arr[b] = t; }

                            void maxHeapify(int arr[], int n, int i) {
                            int largest = i, l = 2*i+1, r = 2*i+2;
                            if (l < n && arr[l] > arr[largest]) largest = l;
                            if (r < n && arr[r] > arr[largest]) largest = r;
                            if (largest != i) { swap(arr, i, largest); maxHeapify(arr, n, largest); }
                            }

                            void insertIntoMaxHeap(int arr[], int* n, int val) {
                            arr[*n] = val; (*n)++;
                            int cur = *n - 1;
                            while (cur > 0 && arr[(cur-1)/2] < arr[cur]) {
                            swap(arr, cur, (cur-1)/2);
                            cur = (cur-1)/2;
                            }
                            }

                            void deleteRoot(int arr[], int* n) {
                            if (*n <= 0) return;
                            arr[0] = arr[*n - 1]; (*n)--;
                            maxHeapify(arr, *n, 0);
                            }

                            int sumLeafNodes(int arr[], int n) {
                            int sum = 0;
                            for (int i = n/2; i < n; i++) sum += arr[i];
                            return sum;
                            }

                            int main() {
                            int arr[MAX_SIZE], n;
                            cin >> n;
                            for (int i=0; i<n; i++) { int v; cin >> v; insertIntoMaxHeap(arr, &n, v); }
                            for (int i=0; i<n; i++) cout << arr[i] << (i<n-1?" ":""); cout << endl;
                            deleteRoot(arr, &n);
                            for (int i=0; i<n; i++) cout << arr[i] << (i<n-1?" ":""); cout << endl;
                            cout << "Sum of leaf nodes: " << sumLeafNodes(arr, n) << endl;
                            return 0;
                            }
💡 Key idea: After deleting the root (max element), leaf nodes are those with indices from n/2 to n-1. Sum them directly.
📥 Input
5
4 3 2 5 6
📤 Output
6 5 2 3 4
5 4 2 3
Sum of leaf nodes: 5

🔄 Heap Sort – Ascending Order

Heap Sort Array Sorting
Problem: Diego sorts candy sweetness levels in non‑decreasing order using Heap Sort.
C++ 
// Heap Sort – Ascending Order
                            #include <iostream>
                            using namespace std;

                            void swap(int* a, int* b) { int t = *a; *a = *b; *b = t; }

                            void heapify(int arr[], int size, int root) {
                            int largest = root, l = 2*root+1, r = 2*root+2;
                            if (l < size && arr[l] > arr[largest]) largest = l;
                            if (r < size && arr[r] > arr[largest]) largest = r;
                            if (largest != root) { swap(&arr[root], &arr[largest]); heapify(arr, size, largest); }
                            }

                            void heapSort(int arr[], int size) {
                            for (int i = size/2 - 1; i >= 0; i--) heapify(arr, size, i);
                            for (int i = size-1; i > 0; i--) {
                            swap(&arr[0], &arr[i]);
                            heapify(arr, i, 0);
                            }
                            }

                            int main() {
                            int n; cin >> n;
                            int arr[100];
                            for (int i=0; i<n; i++) cin >> arr[i];
                            heapSort(arr, n);
                            for (int i=0; i<n; i++) cout << arr[i] << (i<n-1?" ":"");
                            return 0;
                            }
💡 Key idea: Build a max heap, then repeatedly swap the root (largest) with the last element and heapify the reduced heap. Sorts in-place in O(n log n).
📥 Input
5
275 635 987 145 325
📤 Output
145 275 325 635 987

Heap Sort Digits Descending

// Heap Sort – Sort Digits Descending

                            using namespace std;

                            void swap(int* a, int* b) { int t = *a; *a = *b; *b = t; }

                            void heapify(int arr[], int size, int root) {
                            int largest = root, l = 2*root+1, r = 2*root+2;
                            if (l < size && arr[l] > arr[largest]) largest = l;
                            if (r < size && arr[r] > arr[largest]) largest = r;
                            if (largest != root) { swap(&arr[root], &arr[largest]); heapify(arr, size, largest); }

                            void heapSort(int arr[], int size) {
                            for (int i = size/2 - 1; i >= 0; i--) heapify(arr, size, i);

🔢 Prime / Composite – Heap Sort

Heap Sort Prime Composite Array
Problem: Emily classifies numbers as prime or composite, sorts each category using Heap Sort, and displays the results.
C++ 
// Prime / Composite – Heap Sort
                            #include <iostream>
                            using namespace std;

                            bool isPrime(int num) {
                            if (num <= 1) return false;
                            for (int i=2; i*i <= num; i++) if (num % i == 0) return false;
                            return true;
                            }

                            void swap(int* a, int* b) { int t = *a; *a = *b; *b = t; }

                            void heapify(int arr[], int size, int root) {
                            int smallest = root, l = 2*root+1, r = 2*root+2;
                            if (l < size && arr[l] < arr[smallest]) smallest = l;
                            if (r < size && arr[r] < arr[smallest]) smallest = r;
                            if (smallest != root) { swap(&arr[root], &arr[smallest]); heapify(arr, size, smallest); }
                            }

                            void heapSort(int arr[], int size) {
                            for (int i = size/2 - 1; i >= 0; i--) heapify(arr, size, i);
                            for (int i = size-1; i > 0; i--) {
                            swap(&arr[0], &arr[i]);
                            heapify(arr, i, 0);
                            }
                            }

                            int main() {
                            int n; cin >> n;
                            int prime[100], comp[100], pc=0, cc=0;
                            for (int i=0; i<n; i++) {
                            int x; cin >> x;
                            if (isPrime(x)) prime[pc++] = x; else comp[cc++] = x;
                            }
                            heapSort(prime, pc); heapSort(comp, cc);
                            for (int i=0; i<pc; i++) cout << prime[i] << (i<pc-1?" ":""); cout << endl;
                            for (int i=0; i<cc; i++) cout << comp[i] << (i<cc-1?" ":"");
                            return 0;
                            }
💡 Key idea: Use a min‑heap to sort both categories in ascending order. Prime numbers appear first, then composite numbers.
📥 Input
6
2 3 4 5 6 7
📤 Output
2 3 5 7
4 6
